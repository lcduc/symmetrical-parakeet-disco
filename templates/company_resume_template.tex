% Custom commands
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-1pt}}
  }
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-1pt}\item
    \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & {\color{dark-grey}\small #2}\vspace{1pt}\\ % top row of resume entry
      \textit{#3} & {\color{dark-grey} \small #4}\\ % second row of resume entry
    \end{tabular*}\vspace{-4pt}
}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{-4pt}}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{0pt}}

\color{text-grey}

\begin{document}

%----------HEADING----------
\begin{center}
    \textbf{\Huge \VAR{full_name|latex}} \\ \vspace{5pt}
    \small
    \BLOCK{if phone}\faPhone* \texttt{\VAR{phone|latex}} \hspace{1pt} $|$ \hspace{1pt}\BLOCK{endif}
    \faEnvelope \hspace{2pt} \texttt{\VAR{email|latex}}
    \\ \vspace{-3pt}
\end{center}

%-----------EXPERIENCE-----------
\BLOCK{if work_history}
\section{EXPERIENCE}
  \resumeSubHeadingListStart
    \BLOCK{for job in work_history}
    \resumeSubheading
      {\VAR{job.company|latex}}{\VAR{job.start_date|latex} -- \VAR{job.end_date|latex if job.end_date else "Present"}}
      {\VAR{job.title|latex}}{}
      \BLOCK{if job.description}
      \resumeItemListStart
        \BLOCK{for bullet in job.description}
        \resumeItem{\VAR{bullet|latex}}
        \BLOCK{endfor}
      \resumeItemListEnd
      \BLOCK{endif}
    \BLOCK{endfor}
  \resumeSubHeadingListEnd
\BLOCK{endif}

%-----------EDUCATION-----------
\BLOCK{if education}
\section{EDUCATION}
  \resumeSubHeadingListStart
    \BLOCK{for entry in education}
    \resumeSubheading
      {\VAR{entry.institution|latex}}{\BLOCK{if entry.start_date}\VAR{entry.start_date|latex} -- \VAR{entry.end_date|latex if entry.end_date else "Present"}\BLOCK{else}\VAR{entry.end_date|latex if entry.end_date else ""}\BLOCK{endif}}
      {\VAR{entry.degree|latex}\BLOCK{if entry.field_of_study} in \VAR{entry.field_of_study|latex}\BLOCK{endif}}{\VAR{entry.level|latex}}
      \BLOCK{if entry.description}
      \resumeItemListStart
        \BLOCK{for bullet in entry.description}
        \resumeItem{\VAR{bullet|latex}}
        \BLOCK{endfor}
      \resumeItemListEnd
      \BLOCK{endif}
    \BLOCK{endfor}
  \resumeSubHeadingListEnd
\BLOCK{endif}

%-----------SKILLS-----------
\BLOCK{if skills}
\section{SKILLS}
 \begin{itemize}[leftmargin=0in, label={}]
    \small{\item{
     \VAR{skills|map('latex')|join(', ')}
    }}
 \end{itemize}
\BLOCK{endif}

%-----------CERTIFICATIONS-----------
\BLOCK{if certifications}
\section{CERTIFICATIONS}
 \begin{itemize}[leftmargin=0in, label={}]
    \small{\item{
     \VAR{certifications|map('latex')|join(', ')}
    }}
 \end{itemize}
\BLOCK{endif}
